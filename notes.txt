int	ft_rgb(int i, t_data *data)
{
	double	t;
	int		r;
	int		g;
	int		b;

	r = (int)(4 * (1 - t) * t * t * t * 255);
	g = (int)(4 * (1 - t) * (1 - t) * t * t * 255);
	b = (int)(4 * (1 - t) * (1 - t) * t * 255);
	return ((r << 16) | (g << 8) | b);
}
